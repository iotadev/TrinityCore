# Optional static modules use their own source/config trees and the native game target.
# This is a small build convention, not an AzerothCore binary compatibility layer.
function(tc_register_module)
  cmake_parse_arguments(MODULE "" "NAME;CONFIG;LOADER" "SOURCES;INCLUDE_DIRS;TEST_SOURCES;TEST_LIBRARIES" ${ARGN})
  if(NOT MODULE_NAME OR NOT MODULE_SOURCES)
    message(FATAL_ERROR "tc_register_module requires NAME and SOURCES")
  endif()
  target_sources(game PRIVATE ${MODULE_SOURCES})
  target_include_directories(game PRIVATE ${MODULE_INCLUDE_DIRS})
  set_property(GLOBAL APPEND PROPERTY TC_ENABLED_MODULE_NAMES "${MODULE_NAME}")
  if(MODULE_TEST_SOURCES)
    set_property(GLOBAL APPEND PROPERTY TC_MODULE_TEST_SOURCES ${MODULE_TEST_SOURCES})
  endif()
  if(MODULE_TEST_LIBRARIES)
    set_property(GLOBAL APPEND PROPERTY TC_MODULE_TEST_LIBRARIES ${MODULE_TEST_LIBRARIES})
  endif()
  if(MODULE_LOADER)
    if(NOT MODULE_LOADER MATCHES "^[A-Za-z_][A-Za-z0-9_]*$")
      message(FATAL_ERROR "Module LOADER must be a C++ function identifier")
    endif()
    set_property(GLOBAL APPEND PROPERTY TC_MODULE_SCRIPT_LOADERS "${MODULE_LOADER}")
  endif()
  if(MODULE_CONFIG)
    get_filename_component(CONFIG_NAME "${MODULE_CONFIG}" NAME)
    string(REGEX REPLACE "\\.dist$" "" CONFIG_NAME "${CONFIG_NAME}")
    set_property(GLOBAL APPEND PROPERTY TC_MODULE_CONFIG_FILES "${CONFIG_NAME}")
    if(UNIX)
      install(FILES "${MODULE_CONFIG}" DESTINATION "${CONF_DIR}/modules")
    else()
      install(FILES "${MODULE_CONFIG}" DESTINATION "${CMAKE_INSTALL_PREFIX}/modules")
    endif()
  endif()
  message(STATUS "Optional module enabled: ${MODULE_NAME}")
endfunction()

file(GLOB MODULE_BUILD_FILES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*/CMakeLists.txt")
foreach(MODULE_BUILD_FILE IN LISTS MODULE_BUILD_FILES)
  get_filename_component(MODULE_SOURCE_DIR "${MODULE_BUILD_FILE}" DIRECTORY)
  get_filename_component(MODULE_NAME "${MODULE_SOURCE_DIR}" NAME)
  string(TOUPPER "${MODULE_NAME}" MODULE_OPTION)
  string(REPLACE "-" "_" MODULE_OPTION "${MODULE_OPTION}")
  option(MODULE_${MODULE_OPTION} "Build optional ${MODULE_NAME}" ON)
  if(MODULE_${MODULE_OPTION})
    add_subdirectory("${MODULE_SOURCE_DIR}" "${MODULE_NAME}")
  else()
    message(STATUS "Optional module disabled: ${MODULE_NAME}")
  endif()
endforeach()
